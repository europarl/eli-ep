Additionnal constraints

  • If a Work has a Participation with a foreseen participant (epvoc:capacityPerson) and a participant (eli-dl:had_participant_person), then both must refer to the same Person. 
    PREFIX epvoc: <https://data.europarl.europa.eu/def/epvoc#>
PREFIX eli-dl: <http://data.europa.eu/eli/eli-draft-legislation-ontology#>
SELECT $this $person1 $person2
WHERE {
 $this epvoc:workHadParticipation ?participation .
 ?participation epvoc:inCapacityOf ?foreseen_participation .
 ?foreseen_participation epvoc:capacityPerson ?person1 .
 ?participation eli-dl:had_participant_person ?person2 .
 FILTER(?person1 != ?person2)
}
  • The local part of the URI must be identical to the content of dcterms:identifier 
    PREFIX dcterms: <http://purl.org/dc/terms/>
SELECT $this ?localName ?identifier
WHERE {
 $this dcterms:identifier ?identifier .
 # extracts everything after last "/" from the URI
 BIND (REPLACE(STR($this), "^.*/([^/]*)$", "$1") as ?localName)
 FILTER(?localName != ?identifier)
}

Additionnal constraints

  • If the ForeseenCapacity is linked to a Process, then the capacity role must be RAPPORTEUR, RAPPORTEUR_OPINION, RAPPORTEUR_SHADOW_REPORT, RAPPORTEUR_SHADOW_OPINION or MEMBER_NEGOTIATION_INTERINSTITUTIONAL 
    PREFIX epvoc: <https://data.europarl.europa.eu/def/epvoc#>
PREFIX ep-roles: <https://data.europarl.europa.eu/def/ep-roles/>
SELECT $this
WHERE {
 $this epvoc:capacityActivity ?process .
 FILTER(regex(STR(?process), "proc/[0-9]{4}-[0-9]{4}[A-Z]?"))
 $this epvoc:capacityRole ?role .
 FILTER(?role NOT IN (ep-roles:RAPPORTEUR, ep-roles:RAPPORTEUR_OPINION, ep-roles:RAPPORTEUR_SHADOW_REPORT, ep-roles:RAPPORTEUR_SHADOW_OPINION, ep-roles:MEMBER_NEGOTIATION_INTERINSTITUTIONAL))
}
